  function generateProfile(data) {
    const categories = {};

    data.forEach((item) => {
      if (!categories[item.category]) {
        categories[item.category] = {
          correct: 0,
          total: 0,
        };
      }

      categories[item.category].total += 1;

      if (item.correct) {
        categories[item.category].correct += 1;
      }
    });

    const scores = {};

    Object.keys(categories).forEach((category) => {
      const item = categories[category];

      scores[category] = Math.round(
        (item.correct / item.total) * 100
      );
    });

    const totalCorrect = data.filter(
      (item) => item.correct
    ).length;

    const generalScore = Math.round(
      (totalCorrect / data.length) * 100
    );

    let level = 'Iniciante';

    if (generalScore >= 40) {
      level = 'Básico';
    }

    if (generalScore >= 70) {
      level = 'Intermediário';
    }

    if (generalScore >= 90) {
      level = 'Avançado';
    }

    const weakAreas = Object.keys(scores)
      .filter((category) => scores[category] < 70)
      .sort((a, b) => scores[a] - scores[b]);

    const strongAreas = Object.keys(scores)
      .filter((category) => scores[category] >= 70);

    const newProfile = {
      level,
      generalScore,
      scores,
      weakAreas,
      strongAreas,
    };

    setProfile(newProfile);

    setScreen('result');
  }
